cmake_minimum_required(VERSION 3.16)
project(orbo_hardware)
find_package(ament_cmake REQUIRED)
find_package(hardware_interface REQUIRED)
find_package(pluginlib REQUIRED)
find_package(rclcpp REQUIRED)
find_package(rclcpp_lifecycle REQUIRED)
find_package(orbo_embedded_interfaces REQUIRED)
add_library(orbo_hardware SHARED src/system.cpp)
target_compile_features(orbo_hardware PUBLIC cxx_std_17)
target_compile_options(orbo_hardware PRIVATE -Wall -Wextra -Wpedantic)
target_include_directories(orbo_hardware PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
ament_target_dependencies(orbo_hardware hardware_interface pluginlib rclcpp rclcpp_lifecycle orbo_embedded_interfaces)
pluginlib_export_plugin_description_file(hardware_interface orbo_hardware.xml)
install(TARGETS orbo_hardware EXPORT export_orbo_hardware LIBRARY DESTINATION lib ARCHIVE DESTINATION lib RUNTIME DESTINATION bin)
install(DIRECTORY include/ DESTINATION include)
ament_export_targets(export_orbo_hardware HAS_LIBRARY_TARGET)
ament_export_dependencies(hardware_interface pluginlib rclcpp orbo_embedded_interfaces)
ament_package()
